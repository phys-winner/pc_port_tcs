cmake_minimum_required(VERSION 3.15)
project(closing_shift_hlslcc LANGUAGES C CXX)
set(HLSLCC_LIBRARY_SHARED OFF CACHE BOOL "" FORCE)
add_subdirectory("${HLSLCC_SOURCE}" hlslcc)
target_compile_definitions(hlslcc PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(hlslcc_tool main.cpp)
target_compile_features(hlslcc_tool PRIVATE cxx_std_14)
target_include_directories(hlslcc_tool PRIVATE "${HLSLCC_SOURCE}/src")
target_link_libraries(hlslcc_tool PRIVATE hlslcc d3dcompiler)
add_executable(hlslcc_tool_ubo main.cpp)
target_compile_features(hlslcc_tool_ubo PRIVATE cxx_std_14)
target_include_directories(hlslcc_tool_ubo PRIVATE "${HLSLCC_SOURCE}/src")
target_link_libraries(hlslcc_tool_ubo PRIVATE hlslcc d3dcompiler)
